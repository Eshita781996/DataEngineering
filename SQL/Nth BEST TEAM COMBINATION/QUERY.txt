
WITH TeamCTE AS (
    -- Anchor: start with each player
    SELECT 
        CAST(player_id AS VARCHAR(MAX)) AS team,
        player_id,
        score,
        1 AS team_size,
        player_id AS last_id
    FROM Players

    UNION ALL

    -- Recursive: add next player with higher id to avoid duplicates
    SELECT
        CAST(t.team + ',' + CAST(p.player_id AS VARCHAR(10)) AS VARCHAR(MAX)) AS team,
        t.player_id,
        t.score + p.score AS score,
        t.team_size + 1,
        p.player_id AS last_id
    FROM TeamCTE t
    JOIN Players p 
        ON p.player_id > t.last_id
    WHERE t.team_size < 3   -- target team size = 3
)

-- Final selection: only full teams (size = 3)
SELECT 
    ROW_NUMBER() OVER (ORDER BY score DESC) AS rank_num,
    team,
    score AS total_score
FROM TeamCTE
WHERE team_size = 3
ORDER BY score DESC;
